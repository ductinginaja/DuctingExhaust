  <!-- ============ MENGAPA MANUAL ============ -->
  <section class="dk why" id="metode">
    <div class="wrap">
      <div class="shead rv"><span class="lab"><span class="acc-c">SEC. 01</span> / METODE</span><span class="lab">KENAPA BUKAN MESIN</span></div>
      <h2 class="st rv" style="--d:.05s">Tiga mesin yang kami tolak.</h2>
      <p class="why-intro rv" style="--d:.1s">Kami bukan anti-mesin. Kami anti <em>setengah bersih</em> — hasil yang terlihat rapi dari luar, tapi menyimpan bahan bakar di dalam. Buka tiap argumen di bawah.</p>

      <div class="acc rv" style="--d:.15s">
        <div class="acc-item">
          <button class="acc-btn" aria-expanded="false"><span class="acc-no">M.01</span><span class="acc-t">Jet air hanya memindahkan masalah</span><span class="acc-x"><i data-lucide="plus"></i></span></button>
          <div class="acc-body"><div>
            <p>Tekanan air melempar lemak cair ke sambungan flange, celah damper, dan plenum di balik dinding — lalu memadat kembali di titik yang tidak bisa dilihat siapa pun. Bersih di layar monitor, sumber api baru di tempat tersembunyi.</p>
            <span class="acc-note">FIELD NOTE — Kami pernah mengangkat ±2 kg lemak tersangkut dari plenum klien yang "rutin dibersihkan jet" selama tiga tahun.</span>
          </div></div>
        </div>
        <div class="acc-item">
          <button class="acc-btn" aria-expanded="false"><span class="acc-no">M.02</span><span class="acc-t">Uap melelehkan lemak — lalu lemak itu lari</span><span class="acc-x"><i data-lucide="plus"></i></span></button>
          <div class="acc-body"><div>
            <p>Lemak yang dilunakkan steam menetes mengikuti gravitasi: ke housing fan, ke motor, ke titik terendah ducting. Di sana ia mendingin dan mengeras lebih padat dari sebelumnya — tepat di komponen yang paling mahal untuk dibongkar.</p>
          </div></div>
        </div>
        <div class="acc-item">
          <button class="acc-btn" aria-expanded="false"><span class="acc-no">M.03</span><span class="acc-t">Vakum hanya menghisap yang sudah lepas</span><span class="acc-x"><i data-lucide="plus"></i></span></button>
          <div class="acc-body"><div>
            <p>Kerak lemak 3–6 mm menempel pada galvalum seperti cat. Tidak ada hisapan yang memisahkannya. Pemisahan harus mekanis: bilah scraper diseret sejajar permukaan, sikat kawat untuk sisa film, kanebo untuk finishing. Itu pekerjaan tangan — selalu.</p>
          </div></div>
        </div>
        <div class="acc-item">
          <button class="acc-btn" aria-expanded="false"><span class="acc-no">M.04</span><span class="acc-t">Standar kami: logam yang bisa disentuh</span><span class="acc-x"><i data-lucide="plus"></i></span></button>
          <div class="acc-body"><div>
            <p>Acuan kerja kami adalah prinsip NFPA 96: <b>clean to bare metal</b>, residu lemak maksimal ±0,5 mm. Verifikasi bukan lewat video, melainkan telapak tangan dan foto setiap segmen yang terbuka. Kalau panel tidak dibuka, tidak ada yang berhak mengklaim bersih.</p>
            <span class="acc-note">VERIFIKASI — Sentuh. Foto. Timbang lemaknya. Setiap segmen, setiap proyek.</span>
          </div></div>
        </div>
      </div>
    </div>
  </section>

  <section class="proc" id="proses">

          <h2 class="st rv" style="--d:.05s">Enam langkah. Semua terbuka. Semua dilaporkan.</h2>
          <p class="rv" style="--d:.1s">Tidak ada pekerjaan "di dalam sana" yang tidak Anda lihat. Setiap segmen ducting dibuka, dikerok, difoto sebelum-sesudah, dan masuk laporan yang Anda simpan untuk asuransi maupun pemadam kebakaran.</p>
          <div class="note-box rv" style="--d:.15s"><b>DURASI TIPIKAL —</b> restoran skala menengah selesai dalam 1–2 shift malam. Dapur tetap bisa buka pagi berikutnya. Jadwal menyesuaikan jam operasional Anda, bukan sebaliknya.</div>
        </div>

        <div class="steps">
          <div class="step rv"><div class="step-no">01</div><div>
            <h3><i data-lucide="ruler"></i>Survei &amp; peta akses</h3>

            <div class="meta"><span>EST. <b>60–90 MENIT</b></span><span>BIAYA <b>GRATIS</b></span></div>
          </div></div>
          <div class="step rv" style="--d:.05s"><div class="step-no">02</div><div>
            <h3><i data-lucide="shield-check"></i>Proteksi area dapur</h3>
            <p>Alat masuk ditutup film, lantai dilapisi terpal, area kerja dibatasi barrier. Dapur Anda tetap dapur — bukan lokasi proyek berdebu.</p>
            <div class="meta"><span>MELIPUTI <b>HOOD · STOVE · LANTAI</b></span></div>
          </div></div>
          <div class="step rv" style="--d:.1s"><div class="step-no">03</div><div>
            <h3><i data-lucide="wrench"></i>Pembukaan access panel</h3>

            <div class="meta"><span>PANEL BARU <b>± 30 MENIT/TITIK</b></span><span>SEAL <b>GASKET + SCREW</b></span></div>
          </div></div>
          <div class="step rv" style="--d:.15s"><div class="step-no">04</div><div>
            <h3><i data-lucide="brush"></i>Pengerokan manual</h3>
            <p>Inti pekerjaan: scraper bilah, sikat kawat, kanebo. Kerak dipisahkan dari logam, dikumpulkan, lalu ditimbang — bukan dibilas ke drain Anda.</p>
            <div class="meta"><span>PERKAKAS <b>SCRAPER · WIRE BRUSH</b></span><span>OUTPUT <b>LEMAK DITIMBANG</b></span></div>

            <div class="meta"><span>CEK <b>BALANCING · GETARAN</b></span></div>
          </div></div>
          <div class="step rv" style="--d:.25s"><div class="step-no">06</div><div>

const statIO = new IntersectionObserver(es=>{

/* =========================================================
   LAMPIRAN FOTO — pembanding SEBELUM/SESUDAH (slider geser)
   Struktur tiap .ba:
     <img class="ba-img">       pertama = SESUDAH (lapisan bawah)

 $$('.ba').forEach(ba=>{
  const top = $('.ba-top', ba), handle = $('.ba-handle', ba);
  let x = 50, dragging = false, dead = false;

  const apply = ()=>{
    top.style.clipPath = `inset(0 ${100 - x}% 0 0)`;
    handle.style.left = x + '%';
    ba.setAttribute('aria-valuenow', Math.round(x));
  };
  const fromEvent = e=>{
    const r = ba.getBoundingClientRect();
    x = Math.max(0, Math.min(100, (e.clientX - r.left) / r.width * 100));
    apply();
  };

  ba.addEventListener('pointerdown', e=>{
    dead = true;                 // hentikan animasi intro bila user mulai geser
    dragging = true;
    ba.setPointerCapture(e.pointerId);
    fromEvent(e);
  });
  ba.addEventListener('pointermove', e=>{ if(dragging) fromEvent(e); });
  ['pointerup','pointercancel'].forEach(ev=>ba.addEventListener(ev, ()=>dragging=false));

  ba.addEventListener('keydown', e=>{
    const step = e.shiftKey ? 10 : 3;
    if(e.key==='ArrowLeft'){  x = Math.max(0,  x-step); apply(); e.preventDefault(); }
    if(e.key==='ArrowRight'){ x = Math.min(100, x+step); apply(); e.preventDefault(); }
  });

  apply();

  /* animasi "sentuhan" sekali saat pertama terlihat — menandakan bisa digeser */
  const nz = new IntersectionObserver(es=>{
    es.forEach(en=>{
      if(!en.isIntersecting) return;
      nz.disconnect();
      const tween = (from, to, dur, cb)=>{
        if(dead){ x = 50; apply(); return; }
        const s = performance.now();
        (function f(t){
          if(dead){ return; }
          const k = Math.min(1, (t-s)/dur), e2 = .5 - Math.cos(Math.PI*k)/2;
          x = from + (to-from)*e2; apply();
          if(k<1) requestAnimationFrame(f); else cb && cb();
        })(s);
      };
      tween(50, 64, 420, ()=>tween(64, 36, 560, ()=>tween(36, 50, 420)));
    });
  }, {threshold:.55});
  nz.observe(ba);
});
